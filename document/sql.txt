DROP TABLE IF EXISTS books CASCADE;
CREATE TABLE books(
id serial primary key,
title text,
author text
);

DROP TABLE IF EXISTS reviews CASCADE;
CREATE TABLE reviews(
id serial primary key,
book_id integer,
rate integer,
comment text,
FOREIGN KEY(book_id) REFERENCES books(id)
);
